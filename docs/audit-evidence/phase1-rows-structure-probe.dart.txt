import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:balance_tracker/models/transaction.dart';
import 'package:balance_tracker/screens/accounts_screen.dart';
import 'package:balance_tracker/screens/all_transactions_screen.dart';
import 'package:balance_tracker/screens/dashboard_screen.dart';
import 'package:balance_tracker/services/state.dart';
import 'package:balance_tracker/services/monetization_service.dart';
import 'package:balance_tracker/widgets/transaction_tile.dart';

void main() {
  for (final entry in [
    ('dashboard', const DashboardScreen()),
    ('accounts', const AccountsScreen()),
    ('transactions', const AllTransactionsScreen()),
  ]) {
    testWidgets('${entry.$1}: row paint-boundary diagnostic', (tester) async {
      SharedPreferences.setMockInitialValues({
        'tally_tour_accounts_completed': true,
        'tally_tour_dashboard_completed': true,
      });
      AppState.currentUser = null;
      AppState.transactionsNotifier.value = List.generate(
        1000,
        (i) => Transaction(
          id: 'row_probe_$i', title: 'Sample $i', amount: 1,
          date: DateTime(2026, 10, 24), isIncome: i.isEven,
          category: 'Food & Dining', account: 'Cash',
        ),
      );
      AppState.accountsNotifier.value = ['Cash'];
      AppState.loansNotifier.value = [];
      AppState.goalsNotifier.value = [];
      AppState.budgetsNotifier.value = {};
      MonetizationService.resetToLoggedOut();
      MonetizationService.isProUnlockedNotifier.value = true;
      tester.view.physicalSize = const Size(800, 1200);
      tester.view.devicePixelRatio = 1;
      addTearDown(tester.view.resetPhysicalSize);
      addTearDown(tester.view.resetDevicePixelRatio);
      await tester.pumpWidget(MaterialApp(home: entry.$2));
      await tester.pumpAndSettle();
      final vertical = find.byWidgetPredicate(
        (w) => w is Scrollable && w.axisDirection == AxisDirection.down,
      ).first;
      final page = tester.state<ScrollableState>(vertical).position;
      page.jumpTo(entry.$1 == 'dashboard' ? page.maxScrollExtent : 1400);
      await tester.pumpAndSettle();
      final rows = find.byType(TransactionTile);
      expect(rows, findsWidgets);
      final counts = <int>[];
      for (final row in rows.evaluate()) {
        var boundaries = 0;
        row.visitAncestorElements((ancestor) {
          if (ancestor.widget is Scrollable) return false;
          if (ancestor.widget is RepaintBoundary) boundaries++;
          return true;
        });
        counts.add(boundaries);
      }
      debugPrint('${entry.$1}: built rows=${counts.length}; boundaries between each row and Scrollable=$counts');
      final removedScreens = const String.fromEnvironment('REMOVED_SCREENS', defaultValue: 'all');
      final expected = const bool.fromEnvironment('ROWS_REMOVED') &&
              (removedScreens == 'all' || removedScreens.split(',').contains(entry.$1))
          ? (entry.$1 == 'dashboard' ? 1 : 0) : 2;
      expect(counts, everyElement(expected));
    });
  }
}
